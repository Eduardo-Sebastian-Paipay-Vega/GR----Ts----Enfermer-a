\usepackage{soul}

% 1. Macro para citas pendientes o no validadas en el archivo .bib
\providecommand{\todoCita}[1]{%
  \texthl{\textbf{\textcolor{red}{[FALTA CITA REAL: #1]}}}%
}

% 2. Macro para datos empíricos o porcentajes pendientes de verificación
\providecommand{\datoPendiente}[1]{%
  \fcolorbox{red}{yellow!30}{\textbf{\texttt{[DATO PENDIENTE: #1]}}}%
}

% 3. Macro para términos o variables inmutables
\providecommand{\variableInmutable}[1]{%
  \textbf{#1}%
}

% 4. Macro para categorías fenomenológicas cualitativas inmutables
\providecommand{\categoriaInmutable}[1]{%
  \textbf{\textit{#1}}%
}
